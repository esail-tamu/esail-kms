<script>
  document$.subscribe(function () {
    var root = document.getElementById("fnc-quiz");
    if (!root || root.dataset.fncBound) return;
    root.dataset.fncBound = "true";

    var checkBtn = document.getElementById("fnc-check");
    var resetBtn = document.getElementById("fnc-reset");
    var result = document.getElementById("fnc-result");

    function normalize(value) {
      return value.trim().toLowerCase();
    }

    checkBtn.addEventListener("click", function () {
      var questions = root.querySelectorAll(".fnc-q");
      var correctCount = 0;

      questions.forEach(function (q) {
        var type = q.dataset.type;
        var expected = normalize(q.dataset.answer);
        var display = q.dataset.answerDisplay;
        var feedback = q.querySelector(".fnc-feedback");
        var userAnswer = "";

        if (type === "mcq") {
          var checked = q.querySelector("input[type='radio']:checked");
          userAnswer = checked ? normalize(checked.value) : "";
        } else {
          var input = q.querySelector(".fnc-input");
          userAnswer = normalize(input.value);
        }

        var isCorrect = userAnswer !== "" && userAnswer === expected;
        q.classList.remove("fnc-correct", "fnc-incorrect");
        q.classList.add(isCorrect ? "fnc-correct" : "fnc-incorrect");

        if (isCorrect) {
          correctCount++;
          feedback.textContent = "Correct!";
        } else {
          feedback.textContent = "Not quite — correct answer: " + display;
        }
      });

      result.textContent = "You scored " + correctCount + " / " + questions.length + ".";
      result.hidden = false;
    });

    resetBtn.addEventListener("click", function () {
      root.querySelectorAll(".fnc-q").forEach(function (q) {
        q.classList.remove("fnc-correct", "fnc-incorrect");
        q.querySelector(".fnc-feedback").textContent = "";
        var radios = q.querySelectorAll("input[type='radio']");
        radios.forEach(function (r) { r.checked = false; });
        var input = q.querySelector(".fnc-input");
        if (input) input.value = "";
      });
      result.hidden = true;
    });
  });
</script>
